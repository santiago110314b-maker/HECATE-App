    <script>
        // Actualizar Widget de Hora, Fecha y Clima en tiempo real
        function updateWidget() {
            const now = new Date();
            document.getElementById('widget-time').innerText = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            document.getElementById('widget-date').innerText = now.toLocaleDateString([], { day: '2-digit', month: '2-digit' });
        }
        setInterval(updateWidget, 1000);
        updateWidget();

        // Sistema de Chat y Registro
        const chatLog = document.getElementById('chat-log');
        const userInput = document.getElementById('user-input');

        function appendMessage(sender, text) {
            const msgDiv = document.createElement('div');
            msgDiv.classList.add('message', sender === 'Tú' ? 'user' : 'hecate');
            msgDiv.innerText = text;
            chatLog.appendChild(msgDiv);
            chatLog.scrollTop = chatLog.scrollHeight;
        }

        // Motor de Síntesis de Voz (Hécate Habla)
        function speak(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel(); // Detener voz anterior si la hubiera
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'es-ES';
                utterance.rate = 1.05;
                window.speechSynthesis.speak(utterance);
            }
        }

        // Procesador de Comandos Avanzado
        function processCommand(text) {
            appendMessage("Tú", text);
            userInput.value = "";

            let reply = "Comando procesado y ejecutado con éxito, amo.";
            const query = text.toLowerCase();

            if (query.includes("hola") || query.includes("saludos")) {
                reply = "Hola, amo. Todos los protocolos de Hécate están activos y a su servicio.";
            } else if (query.includes("hora")) {
                reply = "Son las " + new Date().toLocaleTimeString() + ", sistemas sincronizados.";
            } else if (query.includes("clima")) {
                reply = "Los sensores reportan condiciones meteorológicas estables en su zona.";
            } else if (query.includes("abre google")) {
                openWeb("https://www.google.com");
                reply = "Abriendo Google sin restricciones.";
            } else if (query.includes("abre youtube")) {
                openWeb("https://www.youtube.com");
                reply = "Abriendo YouTube.";
            } else if (query.includes("busca") || query.includes("navega")) {
                const searchParam = query.replace("busca", "").replace("navega", "").trim();
                openWeb(`https://www.google.com/search?q=${encodeURIComponent(searchParam)}`);
                reply = `Buscando información sobre ${searchParam} en la red.`;
            }

            appendMessage("Hécate", reply);
            speak(reply);
        }

        function handleSendText() {
            const text = userInput.value.trim();
            if (text !== "") {
                processCommand(text);
            }
        }

        userInput.addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                handleSendText();
            }
        });

        // Reconocimiento de Voz (Comandos de Voz)
        let recognition;
        let isListening = false;

        if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRecognition();
            recognition.lang = 'es-ES';
            recognition.continuous = false;
            recognition.interimResults = false;

            recognition.onstart = function() {
                isListening = true;
                document.getElementById('mic-btn').classList.add('listening');
            };

            recognition.onresult = function(event) {
                const speechToText = event.results[0][0].transcript;
                processCommand(speechToText);
            };

            recognition.onerror = function() {
                stopListening();
            };

            recognition.onend = function() {
                stopListening();
            };
        }

        function toggleSpeechRecognition() {
            if (!recognition) {
                alert("Su navegador no soporta el reconocimiento de voz por API web. Use Google Chrome.");
                return;
            }
            if (isListening) {
                recognition.stop();
            } else {
                try {
                    recognition.start();
                } catch(e) {
                    console.error(e);
                }
            }
        }

        function stopListening() {
            isListening = false;
            document.getElementById('mic-btn').classList.remove('listening');
        }

        // Navegador Web Interno Sin Límites
        function openWeb(url) {
            const overlay = document.getElementById('browser-overlay');
            const frame = document.getElementById('browser-frame');
            const label = document.getElementById('browser-url-label');
            
            frame.src = url;
            label.innerText = url;
            overlay.style.display = 'flex';
        }

        function closeWeb() {
            const overlay = document.getElementById('browser-overlay');
            const frame = document.getElementById('browser-frame');
            frame.src = '';
            overlay.style.display = 'none';
        }
    </script>
